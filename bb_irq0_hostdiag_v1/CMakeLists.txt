cmake_minimum_required(VERSION 3.20)
project(bb_irq0_hostdiag_v1 C)
set(CMAKE_C_STANDARD 11)
set(MICRODOS_ROOT "C:/microDOS" CACHE PATH "Existing microDOS checkout")
if(NOT EXISTS "${MICRODOS_ROOT}/include/microdos/runtime.h")
  message(FATAL_ERROR "No microDOS source at ${MICRODOS_ROOT}; pass -DMICRODOS_ROOT=C:/microDOS")
endif()
find_library(MICRODOS_RUNTIME_LIB NAMES microdos_runtime
  PATHS "${MICRODOS_ROOT}/build-host/Release" "${MICRODOS_ROOT}/build-host/Debug"
  NO_DEFAULT_PATH)
if(NOT MICRODOS_RUNTIME_LIB)
  message(FATAL_ERROR "microdos_runtime.lib not found. First: cmake --build C:/microDOS/build-host --config Release --target microdos_runtime")
endif()
add_executable(bb_irq0_roundtrip bb_irq0_roundtrip.c)
target_include_directories(bb_irq0_roundtrip PRIVATE "${MICRODOS_ROOT}/include")
target_link_libraries(bb_irq0_roundtrip PRIVATE "${MICRODOS_RUNTIME_LIB}")
if(MSVC)
  target_compile_options(bb_irq0_roundtrip PRIVATE /W4)
else()
  target_compile_options(bb_irq0_roundtrip PRIVATE -Wall -Wextra -Wpedantic)
endif()
