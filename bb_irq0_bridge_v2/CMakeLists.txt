cmake_minimum_required(VERSION 3.20)
project(bb_irq0_bridge_v2 C)
set(CMAKE_C_STANDARD 11)
set(MICRODOS_ROOT "C:/microDOS" CACHE PATH "Current microDOS root")
set(BLITZ86_ROOT "C:/blitz86_v2" CACHE PATH "Current blitz86 root")
if(NOT EXISTS "${MICRODOS_ROOT}/include/microdos/runtime.h")
  message(FATAL_ERROR "Missing microDOS headers: ${MICRODOS_ROOT}")
endif()
if(NOT EXISTS "${BLITZ86_ROOT}/include/b86.h")
  message(FATAL_ERROR "Missing blitz86 headers: ${BLITZ86_ROOT}")
endif()
if(NOT EXISTS "${CMAKE_CURRENT_BINARY_DIR}/blitz86/interp.c")
  message(FATAL_ERROR "Run prepare_sources.py before CMake configure")
endif()
if(WIN32 AND MSVC)
  set(MICRODOS_RUNTIME_LIB "${MICRODOS_ROOT}/build-host/Release/microdos_runtime.lib")
else()
  find_library(MICRODOS_RUNTIME_LIB NAMES microdos_runtime
    PATHS "${MICRODOS_ROOT}/build-host" "${MICRODOS_ROOT}/build-host/Release" NO_DEFAULT_PATH)
endif()
if(NOT EXISTS "${MICRODOS_RUNTIME_LIB}")
  message(FATAL_ERROR "Missing runtime library: ${MICRODOS_RUNTIME_LIB}")
endif()
add_executable(bb_irq0_bridge bb_irq0_bridge.c "${CMAKE_CURRENT_BINARY_DIR}/blitz86/interp.c")
target_include_directories(bb_irq0_bridge PRIVATE "${MICRODOS_ROOT}/include" "${CMAKE_CURRENT_BINARY_DIR}/blitz86" "${BLITZ86_ROOT}/include")
target_link_libraries(bb_irq0_bridge PRIVATE "${MICRODOS_RUNTIME_LIB}")
if(MSVC)
    target_compile_options(bb_irq0_bridge PRIVATE /W3 /wd4996)
else()
    target_compile_options(bb_irq0_bridge PRIVATE -Wall -Wextra -Wno-unused-function)
endif()
